def mergeTwoLists(l1, l2):
    dummy = ListNode(0)
    cur = dummy

    while l1 is not None and l2 is not None:
        if l1.val <= l2.val:
            cur.next, l1 = l1, l1.next
        else:
            cur.next, l2 = l2, l2.next
        cur = cur.next

    cur.next = l1 if l1 is not None else l2
    return dummy.next


merge2list = mergeTwoLists
